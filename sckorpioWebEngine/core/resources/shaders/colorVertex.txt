#shadervertex
precision mediump float;

attribute vec2 a_vertPosition;
attribute vec3 a_vertColor;
varying vec3 v_fragColor;

void main()
{
    v_fragColor = a_vertColor;
    gl_Position = vec4(a_vertPosition, 0.0, 1.0);
}

#shaderfragment
precision mediump float;

varying vec3 v_fragColor;

void main()
{
    gl_FragColor = vec4(v_fragColor,1.0);
}
